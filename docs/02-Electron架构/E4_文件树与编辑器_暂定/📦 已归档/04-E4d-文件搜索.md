## 一、定位

VS Code 的搜索面板是独立 Viewlet（`searchView.ts`），不是 Explorer 的附属功能。LinkDesk 照抄这个决策——搜索是文件树插件提供的**第二个视图**：

```
图标栏 📁 = 文件树（侧栏）    ← Explorer view
图标栏 🔍 = 搜索面板（侧栏）  ← Search view
两者都是 file-tree 插件提供的侧栏视图
```

**为什么不是另一个插件？** 因为搜索和文件树深度耦合——搜索结果需要 FileTreeModel 的路径解析、EncodingService 的编码检测、FileAssociationService 的打开匹配。它们共享同一个数据模型。

### #109 文件搜索器（~80 行）

```typescript
// FileSearcher.ts
// 🔥 搜索结果缓冲——大文件夹不阻塞 UI

interface SearchOptions {
  query: string;
  include?: GlobPattern[];
  exclude?: GlobPattern[];
  maxResults?: number;
  caseSensitive?: boolean;
  wholeWord?: boolean;
  useRegex?: boolean;
}

class FileSearcher {
  private _options: SearchOptions;
  private _results: SearchResult[] = [];
  private _cancelToken = false;

  async search(): Promise<SearchResult[]> {
    this._cancelToken = false;
    // 1. 遍历所有 workspace root
    for (const root of model.roots) {
      if (this._cancelToken) break;
      await this._searchDir(root.uri);
    }
    return this._results;
  }

  private async _searchDir(dirPath: string): Promise<void> {
    const entries = await FileService.listDir(dirPath);
    for (const entry of entries) {
      if (this._cancelToken) return;
      if (this._results.length >= this._options.maxResults) return;
      if (this._isExcluded(entry)) continue;
      if (entry.isDirectory) { await this._searchDir(entry.path); }
      else { await this._searchFile(entry.path); }
    }
  }

  private async _searchFile(filePath: string): Promise<void> {
    const buffer = await FileService.readBinaryFile(filePath);
    const encoding = EncodingService.detect(buffer);
    const content = EncodingService.decode(buffer, encoding);
    const lines = content.split('\n');
    const fileResults: SearchMatch[] = [];
    for (let i = 0; i < lines.length; i++) {
      const match = this._matchLine(lines[i], i + 1);
      if (match) fileResults.push(match);
    }
    if (fileResults.length > 0) {
      this._results.push({ filePath, matches: fileResults });
    }
  }

  cancel(): void { this._cancelToken = true; }
}

// 🔥 v1 限制：纯 JS 递归搜索。~5,000 文件以下体验良好（<2s），
//    大仓库（10,000+ 文件）会明显变慢。2000 结果上限 + 取消令牌缓解。
//
// 🔥 v2 迁移路径：RipGrep 子进程
//    VS Code 用 RipGrep 做搜索是有原因的——百万行代码秒出结果。
//    Electron main process 可以 spawn rg 子进程 → IPC 流式传结果到渲染进程。
//    接口兼容——FileSearcher 的 search() 签名不变，底层换引擎。
//    v1 不阻塞 v2——API 契约一致。v1 先交付功能，v2 换引擎不改调用方。
```

### #110 搜索替换（~50 行）

```typescript
// SearchReplace.ts
// 🔥 对标 VS Code search.replace
// 用户填入"替换为"文本框 → 每个结果行旁边出现替换按钮

interface ReplaceOptions {
  files: SearchResultFile[];  // 要替换的文件
  from: string;
  to: string;
}

async function replaceAll(options: ReplaceOptions): Promise<{ replaced: number; files: number }> {
  let totalReplaced = 0;
  let filesModified = 0;
  for (const file of options.files) {
    const content = await FileService.readFile(file.filePath);
    const newContent = content.replaceAll(options.from, options.to);
    if (newContent !== content) {
      await FileService.writeFile(file.filePath, newContent);
      totalReplaced += countMatches(content, options.from);
      filesModified++;
    }
  }
  // 文件树监听 FileService.writeFile → CoreEvents.onDidChangeFileSystem → 自动刷新
  return { replaced: totalReplaced, files: filesModified };
}
```
