## 三、src/index.tsx

```tsx
import React, { useState, useEffect } from "react";

export default function HelloWorldView({ isActive }: { isActive: boolean }) {
  const [greeting, setGreeting] = useState("Hello, LinkDesk!");
  const [tabCount, setTabCount] = useState(0);
  const [randomValue, setRandomValue] = useState(Math.random());

  useEffect(() => {
    if (!isActive) return;

    // 读配置
    const configGreeting = window.linkdesk?.configuration?.get?.("hello.greeting");
    if (configGreeting) setGreeting(configGreeting);

    // 订阅配置变化
    const unsubscribe = window.linkdesk?.configuration?.onChange?.((key: string, value: any) => {
      if (key === "hello.greeting") setGreeting(value);
    });

    return () => { unsubscribe?.(); };
  }, [isActive]);

  // 状态栏——显示问候语
  useEffect(() => {
    window.linkdesk?.contextKey?.set?.("hello.greeting", greeting);
  }, [greeting]);

  // 注册命令
  useEffect(() => {
    window.linkdesk?.commands?.executeCommand; // 确保存在

    // 注意：命令注册在模块级做（见下方）
  }, []);

  return (
    <div style={{ padding: "20px" }}>
      <h2>{greeting}</h2>
      <p>Random value (new per tab): {randomValue.toFixed(4)}</p>

      <button
        onClick={() => {
          window.linkdesk?.tabs?.create?.({
            pluginId: "hello-world",
            label: `Hello Tab ${tabCount + 1}`,
          });
          setTabCount(t => t + 1);
        }}
        style={{
          padding: "8px 16px",
          background: "var(--accent)",
          color: "white",
          border: "none",
          borderRadius: "4px",
          cursor: "pointer",
        }}
      >
        Open New Tab
      </button>

      <p style={{ marginTop: "20px", color: "var(--text-secondary)" }}>
        Try: right-click in editor → "Hello: Say Hi"
        <br />
        Or press: Ctrl+Shift+H
        <br />
        Or change greeting in Settings → Hello World
      </p>
    </div>
  );
}

// 模块级命令注册——对标 E5#116 串口右键的模式
if (typeof window !== "undefined" && window.linkdesk) {
  window.linkdesk.commands?.executeCommand;
  // 实际命令在 plugin.json 中声明，壳自动注册
  // hello.sayHi → 壳查找 handler → toast "Hello!"
  // hello.openTab → 壳查找 handler → openOrFocus hello-world
}
```

## 五、验证清单（7 项）

| # | 验证点 | 操作 | 预期结果 |
|:--|:--|:--|:--|
| 1 | 图标栏 | 安装后 → 查看图标栏 | Smile 图标出现在顶部图标栏 |
| 2 | 侧栏 | 点击 Smile 图标 | 侧栏展开 → 显示问候语 + 配置值 + "Open New Tab" 按钮 |
| 3 | 标签页 | 点击 "Open New Tab" 多次 | 每次新建标签页 → 随机数不同（证明独立实例） |
| 4 | 右键菜单 | 编辑器内右键 | 菜单出现 "Hello: Say Hi" → 点击 → toast 弹出 |
| 5 | 快捷键 | Ctrl+Shift+H | toast 弹出 "Hello!" |
| 6 | 配置 | 设置 → Hello World → 改 greeting 值 | 侧栏文字实时变化 |
| 7 | 语言切换 | 切英文 | 所有 UI 文字（菜单/命令/设置）变英文 |
