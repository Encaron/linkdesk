### 1.2 技术实现

```typescript
async function downloadPlugin(downloadUrl: string, pluginId: string) {
  // 1. 下载
  const response = await fetch(downloadUrl);
  const contentLength = Number(response.headers.get("content-length") ?? 0);
  const reader = response.body!.getReader();
  const chunks: Uint8Array[] = [];
  let received = 0;

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    chunks.push(value);
    received += value.length;
    // 更新进度条
    updateProgress(pluginId, received, contentLength);
  }

  // 2. 组装
  const blob = new Blob(chunks);
  const arrayBuffer = await blob.arrayBuffer();

  // 3. 保存到临时目录
  const tmpPath = path.join(appDataDir, "tmp", `${pluginId}.linkdesk-plugin`);
  await window.linkdesk.filesystem.writeBinaryFile(tmpPath, new Uint8Array(arrayBuffer));

  return tmpPath;
}
```

## 二、安装流程

```
下载完成
  ↓
installBundledPlugin(pluginId, tmpPath)
  ↓
1. 校验——读 plugin.json + 验证必填字段
  ↓
2. 解压到 {userData}/plugins/<pluginId>/
  ↓
3. 写 installed-plugins.json
  ↓
4. loadPlugin(pluginId, { source: "bundled" })
  ↓
5. emit "plugin:installed"
  ↓
6. UI 刷新
   - 图标栏出现新图标
   - 侧栏出现新条目
   - 市场页面更新状态
  ↓
7. 清理临时文件
```

## 三、安装状态判断

marketplace 插件渲染时：

```typescript
function getPluginStatus(pluginId: string, marketEntry: MarketEntry): "install" | "installed" | "update" {
  const installed = readInstalledPlugins(); // 从 installed-plugins.json

  if (!installed[pluginId]) return "install";

  const installedVersion = installed[pluginId].version;
  const marketVersion = marketEntry.version;

  if (semverGt(marketVersion, installedVersion)) return "update";
  return "installed";
}
```

## 四、搜索与分类

### 4.1 搜索

```typescript
// 本地过滤——不发网络请求
function searchPlugins(query: string, plugins: MarketEntry[]): MarketEntry[] {
  const q = query.toLowerCase();
  return plugins.filter(p =>
    p.name.toLowerCase().includes(q) ||
    p.description.toLowerCase().includes(q) ||
    p.author.name.toLowerCase().includes(q)
  );
}
```

### 4.2 分类

读取 `marketplace.json` 中每个插件的 `category` 字段：
- "demo" → 示例
- "editor" → 编辑器增强
- "serial" → 串口工具
- "theme" → 主题
- "language" → 语言
- "other" → 其他

## 五、更新通知

### 5.1 检测时机

- 应用启动后 10 秒（给市场数据缓存时间）
- 手动刷新市场页面时

### 5.2 实现

```typescript
async function checkUpdates() {
  const installed = readInstalledPlugins();
  const market = await fetchMarketplace();

  const updates: { pluginId: string; currentVersion: string; newVersion: string }[] = [];

  for (const [pluginId, info] of Object.entries(installed)) {
    const marketEntry = market.plugins.find(p => p.id === pluginId);
    if (marketEntry && semverGt(marketEntry.version, info.version)) {
      updates.push({
        pluginId,
        currentVersion: info.version,
        newVersion: marketEntry.version,
      });
    }
  }

  if (updates.length > 0) {
    // 状态栏通知铃铛
    emitNotification(`${updates.length} 个插件可更新`);
  }
}
```
